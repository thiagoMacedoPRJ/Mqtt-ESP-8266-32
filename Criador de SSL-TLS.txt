1: genrsa -des3 -out ca.key 2048 // Criar uma chave com 2 bytes de criptografia

2: req -new -x509 -days 3652 -key ca.key -out ca.crt // CRIAR DATA DE VALIDADE DA CHAVE
   
3: genrsa -out server.key 2048 // Certificar de que a data de validade tambem foi criptografada

4: req -new -out server.csr -key server.key // Certificado do Server
   
5: x509 -req -in server.csr -CA ca.crt -CAkey ca.key -CAcreateserial -out server.crt -days 366 // Manter Durante 1 ano
